cmake_minimum_required(VERSION 3.21)
project(line_a11y_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui)
enable_testing()
add_executable(test_voice_bubble test_voice_bubble.cpp)
target_include_directories(test_voice_bubble PRIVATE ../bridge/src)
target_link_libraries(test_voice_bubble PRIVATE Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(test_voice_bubble PRIVATE /utf-8)
endif()
add_test(NAME voice_bubble COMMAND test_voice_bubble)

add_executable(test_message_direction test_message_direction.cpp)
target_include_directories(test_message_direction PRIVATE ../bridge/src)
target_link_libraries(test_message_direction PRIVATE Qt6::Core Qt6::Gui)
add_test(NAME message_direction COMMAND test_message_direction)

add_executable(test_native_message_text test_native_message_text.cpp)
target_include_directories(test_native_message_text PRIVATE ../bridge/src)
target_link_libraries(test_native_message_text PRIVATE Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(test_native_message_text PRIVATE /utf-8)
endif()
add_test(NAME native_message_text COMMAND test_native_message_text)
set_tests_properties(native_message_text PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
