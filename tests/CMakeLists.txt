cmake_minimum_required(VERSION 3.21)
project(line_a11y_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui)
enable_testing()
add_executable(test_voice_bubble test_voice_bubble.cpp)
target_include_directories(test_voice_bubble PRIVATE ../bridge/src)
target_link_libraries(test_voice_bubble PRIVATE Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(test_voice_bubble PRIVATE /utf-8)
endif()
add_test(NAME voice_bubble COMMAND test_voice_bubble)
